{{ define "main" }} {{- /* Conteúdo do #main na home: a listagem de posts. Ordenada por num decrescente (número da entrada da crônica), não por data — as datas exibidas são do evento, não da publicação. .Paginate respeita o pagerSize do hugo.toml. A paginação recebe o $paginator (mesmo objeto) p/ render a janela. */ -}} {{- $posts := where site.RegularPages "Type" "posts" -}} {{- $posts = sort $posts "Params.num" "desc" -}} {{- $paginator := .Paginate $posts -}} {{- range $paginator.Pages }} {{ partial "post-card.html" . }} {{- end }} {{ partial "pagination.html" $paginator }} {{ end }}
